@prefix unit: <http://qudt.org/vocab/unit/> .
@prefix quantitykind: <https://qudt.org/vocab/quantitykind/> .

<#SensorMapping> a rr:TriplesMap ;
  rml:logicalSource [
    rml:source "data.csv" ;
    rml:referenceFormulation ql:CSV
  ] ;

  # Core Subject Map
  rr:subjectMap [
  rr:template "http://example.com/observations/{ts_id}/{unixtimestamp}" ;
  rr:class sosa:Observation
] ;

  # Sensor / ts_id
  rr:predicateObjectMap [
    rr:predicate sosa:madeBySensor ;
    rr:objectMap [
      rr:template "http://example.com/waterinfo/{ts_id}" ;
      rr:termType rr:IRI
    ]
  ] ;

  # Timestamp -> sosa:resultTime
  rr:predicateObjectMap [
    rr:predicate sosa:resultTime ;
    rr:objectMap [
      rml:reference "Timestamp" ;
      rr:datatype xsd:dateTime
    ]
  ] ;

  # Value -> sosa:hasSimpleResult
  rr:predicateObjectMap [
    rr:predicate sosa:hasSimpleResult ;
    rr:objectMap [
      rml:reference "Value" ;
      rr:datatype xsd:double
    ]
  ] ;

  # Unit of Measurement -> Constant IRI
  rr:predicateObjectMap [
    rr:predicate qudt:hasUnit ;
    rr:objectMap [
      rr:constant unit:MicroS-PER-CentiM ;
      rr:termType rr:IRI
    ]
  ] ;

  # Observed Property -> Constant IRI
  rr:predicateObjectMap [
    rr:predicate sosa:observedProperty ;
    rr:objectMap [
      rr:constant quantitykind:ElectricConductivity ;
      rr:termType rr:IRI
    ]
  ] .
